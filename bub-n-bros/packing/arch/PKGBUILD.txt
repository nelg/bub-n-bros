# Three packages, which conflict with each other (install one):
#   bubbros-native           the game on the system's Python, with all the
#                            clients, including the native client (Rust)
#   bubbros-server-embedded  the game server as one executable with its own
#                            Python (PyInstaller), and the native client
#   bubbros-play-alone       only the native client, with that server built in
# The build and installation steps are in packing/linux/ (shared with the
# Debian and RPM packages).

pkgbase=bubbros
pkgname=('bubbros-native' 'bubbros-server-embedded' 'bubbros-play-alone')
pkgver=2.4.0
pkgrel=1
pkgdesc="Bubble Bobble clone for up to 10 players"
arch=('x86_64')
license=('MIT' 'LicenseRef-Artistic')
url="https://github.com/nelg/bub-n-bros"
# python-pip: PyInstaller is installed with pip in prepare() (not packaged)
makedepends=('git' 'imagemagick' 'python' 'python-setuptools' 'python-pip'
             'cargo' 'libx11' 'libxext' 'libxrender' 'alsa-lib' 'libglvnd' 'libxi')
options=('!makeflags')
source=("git+https://github.com/nelg/bub-n-bros.git#tag=v$pkgver")
sha256sums=('SKIP')

# libraries of the native client (Rust)
_native_depends=('alsa-lib' 'libglvnd' 'libxi' 'libx11')
_native_optdepends=('pulseaudio-alsa: native client sound through PulseAudio')

_srcdir() {
  echo "$srcdir/bub-n-bros/bub-n-bros"
}

prepare() {
  cd "$(_srcdir)"
  # the native client's Rust dependencies
  export RUSTUP_TOOLCHAIN=stable
  (cd rust-client && cargo fetch --locked --target "$(rustc -vV | sed -n 's/host: //p')")
  # PyInstaller, for the server-embedded and play-alone packages
  python -m venv "$srcdir/pyinstaller-venv"
  "$srcdir/pyinstaller-venv/bin/python" -m pip install --quiet pyinstaller
}

build() {
  cd "$(_srcdir)"
  export RUSTUP_TOOLCHAIN=stable
  export CARGO_TARGET_DIR=target
  PYTHON=python PYINSTALLER_PYTHON="$srcdir/pyinstaller-venv/bin/python" \
    CARGO_FLAGS=--frozen sh packing/linux/build.sh
}

check() {
  cd "$(_srcdir)/rust-client"
  export RUSTUP_TOOLCHAIN=stable
  cargo test --frozen --release
}

_licenses() {
  install -Dm644 -t "$pkgdir/usr/share/licenses/$pkgname" LICENSE.txt artistic.txt
}

package_bubbros-native() {
  pkgdesc="Bubble Bobble clone for up to 10 players (on the system's Python, all clients)"
  # python + pygame: server, web launcher and Python client
  # libx11/libxext/libxrender: X display driver of the Python client
  depends=('python' 'python-pygame' 'libxext' 'libxrender' "${_native_depends[@]}")
  optdepends=('python-gobject: GTK 3 frontend for the client'
              'gtk3: GTK 3 frontend for the client'
              'python-cairo: GTK 3 frontend for the client'
              'java-runtime: Java applet frontend for the client'
              "${_native_optdepends[@]}")
  provides=('bubbros')
  conflicts=('bubbros' 'bubbros-server-embedded' 'bubbros-play-alone')
  replaces=('bubbros')
  install=bubbros.install
  cd "$(_srcdir)"
  PYTHON=python sh packing/linux/install.sh native "$pkgdir"
  _licenses
}

package_bubbros-server-embedded() {
  pkgdesc="Bubble Bobble clone for up to 10 players (server with its own Python, native client)"
  depends=('zlib' "${_native_depends[@]}")
  optdepends=("${_native_optdepends[@]}")
  provides=('bubbros')
  conflicts=('bubbros' 'bubbros-native' 'bubbros-play-alone')
  cd "$(_srcdir)"
  sh packing/linux/install.sh server-embedded "$pkgdir"
  _licenses
}

package_bubbros-play-alone() {
  pkgdesc="Bubble Bobble clone for up to 10 players (native client with a built-in server)"
  depends=('zlib' "${_native_depends[@]}")
  optdepends=("${_native_optdepends[@]}")
  provides=('bubbros')
  conflicts=('bubbros' 'bubbros-native' 'bubbros-server-embedded')
  cd "$(_srcdir)"
  sh packing/linux/install.sh play-alone "$pkgdir"
  _licenses
}
